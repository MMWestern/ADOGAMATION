They are not database // layers — their visibility is stored locally — but they belong in the same // panel so the author can hide a whole column without hunting for a setting. var structuralCols = [ { key: "chapters", label: "Chapters", color: "#3b82f6" }, { key: "scenes", label: "Scenes", color: "#10b981" } ]; structuralCols.forEach(function (col) { var hidden = sceGetColHidden(col.key); html += "
"; html += "
"; html += "" + escapeHtml(col.label) + ""; html += "
"; html += ""; html += ""; html += ""; html += "
"; html += "
"; }); layers.forEach(function (layer) { // The spine is pinned beside the states in the timeline, so it is not // offered here. Listing it would read as a second story layer to toggle. if (sceIsSpineLayer(layer)) return; html += "
"; html += "
"; html += "" + escapeHtml(layer.name) + ""; html += "
"; html += ""; html += ""; html += ""; html += ""; html += "
"; html += "
"; }); html += ""; container.innerHTML = html; // Wire layer controls wireSCELayerControls(); } function wireSCELayerControls() { // Structural column hide/show (chapters, scenes) document.querySelectorAll(".sce-col-hide-btn").forEach(function (btn) { btn.addEventListener("click", function () { var colKey = btn.getAttribute("data-col-key"); sceSetColHidden(colKey, !sceGetColHidden(colKey)); renderSCETimeline(); renderSCELayerControls(); }); }); // Structural column solo — hides everything else document.querySelectorAll(".sce-col-solo-btn").forEach(function (btn) { btn.addEventListener("click", function () { var colKey = btn.getAttribute("data-col-key"); var soloing = !sceGetColSolo(colKey); // Solo this column: hide all other structural cols and all story layers. ["chapters", "scenes"].forEach(function (k) { if (k !== colKey) sceSetColHidden(k, soloing); }); _sceState.layers.forEach(function (l) { if (sceIsSpineLayer(l)) return; l.is_hidden = soloing; saveSCELayer(l); }); sceSetColSolo(colKey, soloing); renderSCETimeline(); renderSCELayerControls(); }); }); // Structural column lock document.querySelectorAll(".sce-col-lock-btn").forEach(function (btn) { btn.addEventListener("click", function () { var colKey = btn.getAttribute("data-col-key"); sceSetColLocked(colKey, !sceGetColLocked(colKey)); renderSCELayerControls(); }); }); // Hide buttons document.querySelectorAll(".sce-layer-hide-btn").forEach(function (btn) { btn.addEventListener("click", function () { var layerId = Number(btn.getAttribute("data-layer-id")); toggleSCELayerHidden(layerId); }); }); // Solo buttons document.querySelectorAll(".sce-layer-solo-btn").forEach(function (btn) { btn.addEventListener("click", function () { var layerId = Number(btn.getAttribute("data-layer-id")); toggleSCELayerSolo(layerId); }); }); // Lock buttons document.querySelectorAll(".sce-layer-lock-btn").forEach(function (btn) { btn.addEventListener("click", function () { var layerId = Number(btn.getAttribute("data-layer-id")); toggleSCELayerLocked(layerId); }); }); // Delete buttons document.querySelectorAll(".sce-layer-delete-btn").forEach(function (btn) { btn.addEventListener("click", function () { var layerId = Number(btn.getAttribute("data-layer-id")); deleteSCELayer(layerId); }); }); // Add layer button var addBtn = document.getElementById("sceAddLayerBtn"); if (addBtn && !addBtn._wired) { addBtn._wired = true; addBtn.addEventListener("click", function () { addSCELayer(); }); } } function toggleSCELayerHidden(layerId) { var layer = _sceState.layers.find(function (l) { return Number(l.id) === layerId; }); if (!layer) return; layer.is_hidden = !layer.is_hidden; // If hiding a solo layer, also unsolo it if (layer.is_hidden && layer.is_solo) layer.is_solo = false; saveSCELayer(layer); renderSCETimeline(); renderSCELayerControls(); } function toggleSCELayerSolo(layerId) { var layer = _sceState.layers.find(function (l) { return Number(l.id) === layerId; }); if (!layer) return; layer.is_solo = !layer.is_solo; layer.is_hidden = false; // unsolo implies unhide saveSCELayer(layer); renderSCETimeline(); renderSCELayerControls(); } function toggleSCELayerLocked(layerId) { var layer = _sceState.layers.find(function (l) { return Number(l.id) === layerId; }); if (!layer) return; layer.is_locked = !layer.is_locked; saveSCELayer(layer); renderSCELayerControls(); } function deleteSCELayer(layerId) { if (sceMutationBlocked("delete layer")) return; var layer = _sceState.layers.find(function (l) { return Number(l.id) === layerId; }); if (!layer) return; // Count blocks on this layer var blocksOnLayer = _sceState.blocks.filter(function (b) { return Number(b.layer_id) === layerId; }); var msg = "Delete layer '" + layer.name + "'?"; if (blocksOnLayer.length) msg += " This will also delete " + blocksOnLayer.length + " block(s)."; if (!confirm(msg)) return; supabaseRunner .withSuccessHandler(function () { // Remove layer from state _sceState.layers = _sceState.layers.filter(function (l) { return Number(l.id) !== layerId; }); // Remove blocks on this layer _sceState.blocks = _sceState.blocks.filter(function (b) { return Number(b.layer_id) !== layerId; }); renderSCETimeline(); renderSCELayerControls(); renderSCEInspector(); }) .withFailureHandler(function (err) { console.error("[SCE] Failed to delete layer:", err); }) .sceDeleteLayer(layerId); } function addSCELayer() { if (sceMutationBlocked("add layer")) return; var name = prompt("Layer name:"); if (!name) return; var compositionId = _sceState.composition ? _sceState.composition.id : 0; if (!compositionId) return; var payload = { composition_id: compositionId, name: name, color: "#64748b", position: _sceState.layers.length }; supabaseRunner .withSuccessHandler(function (json) { var result = typeof json === 'string' ? JSON.parse(json) : json; if (result.layer) { _sceState.layers.push(result.layer); renderSCETimeline(); renderSCELayerControls(); } }) .withFailureHandler(function (err) { console.error("[SCE] Failed to add layer:", err); }) .sceSaveLayer(JSON.stringify(payload)); } function saveSCELayer(layer) { if (sceMutationBlocked("layer edit")) return; supabaseRunner .withSuccessHandler(function () {}) .withFailureHandler(function (err) { console.error("[SCE] Failed to save layer:", err); }) .sceSaveLayer(JSON.stringify(layer)); }
