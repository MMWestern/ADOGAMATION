// ============================================================ // Story Composition Editor — Asset Bin // ============================================================ // An idea belongs to one book, or is shared across the series when its // book_id is null. The filter scope is "book" (this book plus shared), // "shared", "all", or "book:" for one specific book in the series. function sceCurrentBookId() { return sceBookId(getSelectedProject()); } function sceAssetBookOptions() { var seriesId = _sceState.seriesId || 0; var project = getSelectedProject(); var currentBookId = sceCurrentBookId(); var books = []; if (typeof getProjectsForSeries === "function" && project) { books = getProjectsForSeries(String(project.series || "").trim()) || []; } return { currentBookId: currentBookId, seriesId: seriesId, books: books.filter(function (b) { return sceBookId(b) > 0; }) }; } function sceAssetMatchesScope(asset, scope) { var assetBookId = Number(asset && asset.book_id || 0); if (scope === "all") return true; if (scope === "shared") return !assetBookId; if (scope === "book") { // This book shows its own ideas plus anything shared with the series. var opts = sceAssetBookOptions(); return !assetBookId || assetBookId === opts.currentBookId; } if (scope.indexOf("book:") === 0) { return assetBookId === Number(scope.slice(5) || 0); } // Unrecognised scope must not expose other books' ideas. return !assetBookId || assetBookId === sceAssetBookOptions().currentBookId; } function sceFilteredAssets() { var scope = String(_sceState.assetScope || "book"); var q = String(_sceState.assetQuery || "").toLowerCase().trim(); return (_sceState.assets || []).filter(function (asset) { if (!sceAssetMatchesScope(asset, scope)) return false; if (!q) return true; return [asset.name, asset.description, asset.what_changes, asset.what_produces, asset.notes] .map(function (value) { return String(value || "").toLowerCase(); }) .some(function (value) { return value.indexOf(q) !== -1; }); }); } // sce_assets.book_id is a foreign key to projects.id, which reaches the // client as rowNumber. project_id can be a legacy string like "P1", so it // must never be used as the book key or it resolves to NaN. function sceBookId(book) { var candidates = [book && book.rowNumber, book && book.id, book && book.project_id]; for (var i = 0; i < candidates.length; i++) { var n = Number(candidates[i]); if (n > 0) return n; } return 0; } // Books are named by title. The book number stays in the tooltip so the two // identifiers never get confused for each other. function sceBookLabel(book, currentBookId) { var title = String(book && book.title || "").trim(); var label = title || "Untitled book"; if (currentBookId && sceBookId(book) === Number(currentBookId)) label += " (current)"; return label; } function sceBookTitle(book) { return String(book && book.title || "").trim() || "Untitled book"; } function renderSCEAssetBookFilter() { var select = document.getElementById("sceAssetBookFilter"); if (!select) return; var opts = sceAssetBookOptions(); var current = String(_sceState.assetScope || "book"); // The current book carries the default scope (own plus shared), so it is // listed once, by name and marked. Listing it again would give two // entries with the same name meaning different things. var currentBook = opts.books.filter(function (b) { return sceBookId(b) === opts.currentBookId; })[0]; var currentProject = getSelectedProject(); var currentLabel = currentBook ? sceBookLabel(currentBook, opts.currentBookId) : (sceBookTitle(currentProject) + " (current)"); var html = ""; html += ""; html += ""; opts.books.forEach(function (book) { var bookId = sceBookId(book); if (!bookId || bookId === opts.currentBookId) return; html += ""; }); select.innerHTML = html; if (!select.querySelector("option[value='" + current.replace(/'/g, "\\'") + "']")) { current = "book"; _sceState.assetScope = "book"; } select.value = current; } // Surfaces a missing migration in the panel rather than only the console, so // a silently-unsaved book tag cannot look like a successful save. function sceCheckAssetBookScope() { var notice = document.getElementById("sceAssetScopeNotice"); if (typeof supabaseRunner === "undefined") return; supabaseRunner .withSuccessHandler(function (json) { var result = typeof json === 'string' ? JSON.parse(json) : json; if (!notice) return; notice.hidden = !!(result && result.ready); }) .withFailureHandler(function () { if (notice) notice.hidden = true; }) .sceAssetBookScopeReady(); } function renderSCEAssetBin() { var list = document.getElementById("sceAssetBinList"); if (!list) return; renderSCEAssetBookFilter(); sceCheckAssetBookScope(); var assets = _sceState.assets || []; var filtered = sceFilteredAssets(); var countEl = document.getElementById("sceAssetCount"); if (countEl) { countEl.textContent = filtered.length === assets.length ? assets.length + " idea" + (assets.length === 1 ? "" : "s") : filtered.length + " of " + assets.length; } if (!assets.length) { list.innerHTML = "
No ideas yet. Click '+ Add' to create one.
"; return; } if (!filtered.length) { list.innerHTML = "
No ideas match this filter.
"; return; } var currentBookId = sceCurrentBookId(); var bookTitles = {}; sceAssetBookOptions().books.forEach(function (book) { var bookId = sceBookId(book); if (bookId) bookTitles[bookId] = sceBookLabel(book, 0); }); var html = ""; filtered.forEach(function (asset) { var assetBookId = Number(asset.book_id || 0); var html2 = "
"; html2 += "
"; html2 += "
" + escapeHtml(asset.name || "Untitled") + "
"; if (assetBookId) { var tag = bookTitles[assetBookId] || "Book"; html2 += "
" + escapeHtml(tag) + "
"; } else { html2 += "
Shared
"; } html2 += "
"; if (asset.what_changes) { html2 += "
" + escapeHtml(String(asset.what_changes).substring(0, 60)) + "
"; } html2 += "
"; html += html2; }); list.innerHTML = html; if (_sceState.selectedAssetId) { var selected = list.querySelector(".sce-asset-item[data-asset-id='" + Number(_sceState.selectedAssetId) + "']"); if (selected) selected.classList.add("is-selected"); } // Wire asset clicks and drag list.querySelectorAll(".sce-asset-item").forEach(function (item) { item.addEventListener("click", function () { var assetId = Number(item.getAttribute("data-asset-id")); selectSCEAsset(assetId); }); item.addEventListener("dragstart", function (e) { var assetId = item.getAttribute("data-asset-id"); e.dataTransfer.setData("text/plain", JSON.stringify({ type: "asset", assetId: assetId })); e.dataTransfer.effectAllowed = "copy"; }); }); } function selectSCEAsset(assetId) { _sceState.selectedAssetId = assetId; _sceState.selectedBlockId = null; // Update visual selection document.querySelectorAll(".sce-asset-item").forEach(function (el) { el.classList.toggle("is-selected", Number(el.getAttribute("data-asset-id")) === assetId); }); // Update inspector var asset = _sceState.assets.find(function (a) { return Number(a.id) === assetId; }); if (asset) { renderSCEInspectorAsset(asset); } } function showSCEAssetEditor(asset, seriesId) { if (sceMutationBlocked(asset ? "edit asset" : "add asset")) return; var isNew = !asset; var name = prompt(isNew ? "Idea name:" : "Idea name:", asset ? asset.name : ""); if (name === null) return; var payload = { series_id: seriesId, // New ideas belong to the book being written. Editing keeps the idea's // existing scope, so renaming never moves it between books. book_id: isNew ? sceCurrentBookId() : Number(asset.book_id || 0) || null, name: name || "Untitled Idea" }; if (asset) payload.id = asset.id; supabaseRunner .withSuccessHandler(function (json) { var result = typeof json === 'string' ? JSON.parse(json) : json; if (result.asset) { if (isNew) { _sceState.assets.unshift(result.asset); } else { var idx = _sceState.assets.findIndex(function (a) { return Number(a.id) === Number(result.asset.id); }); if (idx >= 0) _sceState.assets[idx] = result.asset; } renderSCEAssetBin(); } }) .withFailureHandler(function (err) { console.error("[SCE] Failed to save asset:", err); }) .sceSaveAsset(JSON.stringify(payload)); } function createSCEBlockFromAsset(assetId, layerId, statePosition, sceneId) { if (sceMutationBlocked("add block")) return; var asset = _sceState.assets.find(function (a) { return Number(a.id) === Number(assetId); }); if (!asset) return; var compositionId = _sceState.composition ? _sceState.composition.id : 0; if (!compositionId) return; var payload = { composition_id: compositionId, asset_id: asset.id, layer_id: layerId, start_state: statePosition, end_state: statePosition, scene_id: sceneId ? Number(sceneId) : null, // Lands at the end of the cell it is dropped into, so a new block never // jumps above one the author already placed there. vertical_pos: sceNextVerticalPos(layerId, statePosition, sceneId), short_title: asset.name || "Untitled", description: asset.description || "", what_happens: asset.what_changes || "" }; // Blocks created from an asset have no catalogue type, so they stay out of // Story Mix. Provenance is only ever set by the catalogue path. payload.source_catalogue_id = null; payload.source_catalogue_name = null; payload.source_category = null; supabaseRunner .withSuccessHandler(function (json) { var result = typeof json === 'string' ? JSON.parse(json) : json; if (result.block) { _sceState.blocks.push(result.block); scePushUndo({ type: "create_block_from_asset", blockId: Number(result.block.id), undo: function (done) { var id = Number(result.block.id); _sceState.blocks = _sceState.blocks.filter(function (b) { return Number(b.id) !== id; }); if (Number(_sceState.selectedBlockId) === id) _sceState.selectedBlockId = null; supabaseRunner .withSuccessHandler(function () { renderSCETimeline(); if (done) done(); }) .withFailureHandler(function (err) { console.error("[SCE] Failed to undo block creation:", err); if (done) done(true); }) .sceDeleteBlock(id); }, redo: function (done) { var block = Object.assign({}, result.block); delete block.id; supabaseRunner .withSuccessHandler(function () { renderSCETimeline(); if (done) done(); }) .withFailureHandler(function (err) { console.error("[SCE] Failed to redo block creation:", err); if (done) done(true); }) .sceSaveBlock(JSON.stringify(block)); } }); renderSCEBlocks(); selectSCEBlock(result.block.id); } }) .withFailureHandler(function (err) { console.error("[SCE] Failed to create block:", err); }) .sceSaveBlock(JSON.stringify(payload)); }
