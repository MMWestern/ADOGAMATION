// ============================================================ // MASTER CODEX v2 — Dual-write helpers // These functions write to v2 tables alongside legacy writes. // Controlled by CODEX_V2_DUAL_WRITE flag. // ============================================================ /** * Record an entity appearance when an entity is linked to a project/book. * Called from saveCodexEntityProjects and entity save paths. */ function codexV2RecordAppearance(seriesId, entityId, bookId, appearanceType) { if (typeof CODEX_V2_DUAL_WRITE === "undefined" || !CODEX_V2_DUAL_WRITE) return; if (!seriesId || !entityId || !bookId) return; var payload = { entity_id: entityId, series_id: seriesId, book_id: bookId, appearance_type: appearanceType || "appears", source: "dual_write" }; supabaseRunner .withSuccessHandler(function () {}) .withFailureHandler(function (err) { console.warn("[codexV2] appearance dual-write failed:", err); }) .saveCodexEntityAppearance(seriesId, JSON.stringify(payload)); } /** * Record a state change when an entity property changes. * Called from entity save paths when key properties change. */ function codexV2RecordState(seriesId, entityId, propertyKey, valueText, valueNumber, bookId, chapterId) { if (typeof CODEX_V2_DUAL_WRITE === "undefined" || !CODEX_V2_DUAL_WRITE) return; if (!seriesId || !entityId || !propertyKey) return; var payload = { subject_entity_id: entityId, property_key: propertyKey, value_text: valueText || null, value_number: valueNumber != null ? valueNumber : null, series_id: seriesId, book_id: bookId || null, chapter_id: chapterId || null, state_status: "draft", source: "dual_write" }; supabaseRunner .withSuccessHandler(function () {}) .withFailureHandler(function (err) { console.warn("[codexV2] state dual-write failed:", err); }) .saveCodexEntityState(seriesId, JSON.stringify(payload)); } /** * Record a progression event when a significant entity change occurs. * Called from entity save paths for status changes, class changes, etc. */ function codexV2RecordEvent(seriesId, entityId, eventType, reason, bookId, chapterId) { if (typeof CODEX_V2_DUAL_WRITE === "undefined" || !CODEX_V2_DUAL_WRITE) return; if (!seriesId || !entityId || !eventType) return; var payload = { subject_entity_id: entityId, event_type: eventType, series_id: seriesId, book_id: bookId || null, chapter_id: chapterId || null, reason: reason || null, canon_status: "draft", source: "dual_write" }; supabaseRunner .withSuccessHandler(function () {}) .withFailureHandler(function (err) { console.warn("[codexV2] event dual-write failed:", err); }) .saveCodexProgressionEvent(seriesId, JSON.stringify(payload)); } /** * Dual-write connection with context fields. * Called from saveCodexConnection when book/chapter context is available. */ function codexV2UpdateConnectionContext(connectionId, bookId, chapterId, status, notes) { if (typeof CODEX_V2_DUAL_WRITE === "undefined" || !CODEX_V2_DUAL_WRITE) return; if (!connectionId) return; // The connection is already saved by the caller — we just need to update the new context fields // This is done via a direct update since the connection already exists var updates = {}; if (bookId) updates.book_id = bookId; if (chapterId) updates.chapter_id = chapterId; if (status) updates.context_status = status; if (notes) updates.notes = notes; if (Object.keys(updates).length === 0) return; supabaseRunner .withSuccessHandler(function () {}) .withFailureHandler(function (err) { console.warn("[codexV2] connection context dual-write failed:", err); }) .updateCodexConnectionContext(connectionId, JSON.stringify(updates)); } /** * Backfill appearances from existing entity_projects. * Only backfills unambiguous data: entity linked to a project = entity appears in that book. * Returns a Promise that resolves with the count of backfilled records. */ function codexV2BackfillAppearances(seriesId) { return new Promise(function (resolve, reject) { if (!seriesId) { reject(new Error("Series id is required.")); return; } console.log("[codexV2] Starting appearances backfill for series", seriesId); // Step 1: Load all entities for this series with their project links supabaseRunner .withSuccessHandler(function (json) { try { var result = JSON.parse(json); if (!result.ok || !result.entities) { reject(new Error("Failed to load entities.")); return; } var entities = result.entities; console.log("[codexV2] Found", entities.length, "entities to backfill"); // Step 2: For each entity with project links, create appearance records var backfillCount = 0; var promises = []; entities.forEach(function (entity) { var entityId = Number(entity.id); var projectLinks = entity.projectLinks || []; projectLinks.forEach(function (link) { var pid = Number(link.project_id || link.projectId || 0); if (pid) { backfillCount++; promises.push(new Promise(function (res) { var payload = { entity_id: entityId, series_id: seriesId, book_id: pid, appearance_type: "appears", source: "backfill" }; supabaseRunner .withSuccessHandler(function () { res(true); }) .withFailureHandler(function (err) { console.warn("[codexV2] backfill failed for entity", entityId, "book", pid, err); res(false); }) .saveCodexEntityAppearance(seriesId, JSON.stringify(payload)); })); } }); }); // Step 3: Wait for all backfill operations to complete Promise.all(promises).then(function (results) { var successCount = results.filter(function (r) { return r; }).length; console.log("[codexV2] Backfill complete:", successCount, "of", backfillCount, "records created"); resolve({ total: backfillCount, success: successCount }); }); } catch (e) { reject(e); } }) .withFailureHandler(function (err) { reject(err); }) .listCodexEntities(seriesId); }); }
