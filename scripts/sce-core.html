if (!seriesId || !bookId) { var root = document.getElementById("sceRoot"); if (root) root.innerHTML = "
Select a book project to use the Story Composition Editor.
"; return; } _sceState.initialized = true; populateSCEProjectInfo(project); loadSCEData(seriesId, bookId); } function populateSCEProjectInfo(project) { var img = document.getElementById("sceProjectImage"); var title = document.getElementById("sceProjectTitle"); var series = document.getElementById("sceProjectSeries"); if (img) { if (typeof getProjectDisplayImageUrl === "function") { img.src = getProjectDisplayImageUrl(project); } else { img.src = project.image_url || project.series_image_url || ""; } img.alt = project.title ? project.title + " cover" : ""; } if (title) title.textContent = project.title || "Untitled"; if (series) series.textContent = project.series_name || project.series || ""; } function loadSCEData(seriesId, bookId) { supabaseRunner .withSuccessHandler(function (json) { var result = typeof json === 'string' ?
