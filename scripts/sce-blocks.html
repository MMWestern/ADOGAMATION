// ============================================================ // Story Composition Editor — Block Drag/Drop and Resize // ============================================================ function wireSCEBlockInteractions() { var timelineBody = document.getElementById("sceTimelineBody"); if (!timelineBody) return; // Allow drops on layer cells timelineBody.addEventListener("dragover", function (e) { var cell = e.target.closest(".sce-layer-cell"); if (cell) { e.preventDefault(); e.dataTransfer.dropEffect = "copy"; cell.classList.add("sce-drop-target"); } }); timelineBody.addEventListener("dragleave", function (e) { var cell = e.target.closest(".sce-layer-cell"); if (cell) { cell.classList.remove("sce-drop-target"); } }); timelineBody.addEventListener("drop", function (e) { var cell = e.target.closest(".sce-layer-cell"); if (!cell) return; cell.classList.remove("sce-drop-target"); e.preventDefault(); var data; try { data = JSON.parse(e.dataTransfer.getData("text/plain")); } catch (err) { return; } if (data.type === "asset") { var layerId = Number(cell.getAttribute("data-layer-id")); var state = cell.getAttribute("data-state"); if (state === "parking" || state === "later") return; var statePosition = Number(state); createSCEBlockFromAsset(data.assetId, layerId, statePosition); } }); // Click on empty cell to deselect timelineBody.addEventListener("click", function (e) { if (e.target.closest(".sce-block")) return; _sceState.selectedBlockId = null; _sceState.selectedAssetId = null; document.querySelectorAll(".sce-block.is-selected").forEach(function (el) { el.classList.remove("is-selected"); }); renderSCEInspector(); }); // Wire block drag (for moving blocks) wireSCEBlockDrag(); } function wireSCEBlockDrag() { // Use mousedown/mousemove/mouseup for block dragging var timelineBody = document.getElementById("sceTimelineBody"); if (!timelineBody) return; var dragState = null; timelineBody.addEventListener("mousedown", function (e) { var block = e.target.closest(".sce-block"); if (!block) return; if (e.target.closest(".sce-block-resize-handle")) return; if (e.target.closest(".sce-block-connect-node")) return; var blockId = Number(block.getAttribute("data-block-id")); var blockData = _sceState.blocks.find(function (b) { return Number(b.id) === blockId; }); if (!blockData) return; // Check if layer is locked var layer = _sceState.layers.find(function (l) { return Number(l.id) === Number(blockData.layer_id); }); if (layer && layer.is_locked) return; var rect = block.getBoundingClientRect(); dragState = { blockId: blockId, startX: e.clientX, startY: e.clientY, blockEl: block, blockRect: rect, moved: false, ghost: null }; e.preventDefault(); }); document.addEventListener("mousemove", function (e) { if (!dragState) return; if (!dragState.moved) { // First move: create ghost and hide original dragState.moved = true; var ghost = document.createElement("div"); ghost.className = "sce-block-drag-ghost"; ghost.style.cssText = "position:fixed;left:" + dragState.blockRect.left + "px;top:" + dragState.blockRect.top + "px;width:" + dragState.blockRect.width + "px;height:" + dragState.blockRect.height + "px;background:" + (dragState.blockEl.style.getPropertyValue("--layer-color") || "#3b82f6") + ";border-radius:4px;padding:4px 8px;z-index:99999;pointer-events:none;opacity:0.8;display:flex;align-items:center;"; var title = dragState.blockEl.querySelector(".sce-block-title"); ghost.innerHTML = title ? "" + title.innerHTML + "" : ""; document.body.appendChild(ghost); dragState.ghost = ghost; dragState.blockEl.style.opacity = "0.3"; } // Move ghost if (dragState.ghost) { dragState.ghost.style.left = (e.clientX - dragState.startX + dragState.blockRect.left) + "px"; dragState.ghost.style.top = (e.clientY - dragState.startY + dragState.blockRect.top) + "px"; } // Auto-scroll timeline when near edges var timeline = document.getElementById("sceTimeline"); if (timeline) { var wrapRect = timeline.getBoundingClientRect(); var edgeZone = 40; var scrollSpeed = 15; if (e.clientX < wrapRect.left + edgeZone) { timeline.scrollLeft -= scrollSpeed; } else if (e.clientX > wrapRect.right - edgeZone) { timeline.scrollLeft += scrollSpeed; } } }); document.addEventListener("mouseup", function (e) { if (!dragState) return; // Remove ghost if (dragState.ghost) { dragState.ghost.remove(); } dragState.blockEl.style.opacity = ""; if (dragState.moved) { // Find drop target var target = document.elementFromPoint(e.clientX, e.clientY); var cell = target ? target.closest(".sce-layer-cell") : null; if (cell) { var newLayerId = Number(cell.getAttribute("data-layer-id")); var newState = cell.getAttribute("data-state"); if (newState !== "parking" && newState !== "later") { var newPosition = Number(newState); moveSCEBlock(dragState.blockId, newLayerId, newPosition); } } } dragState = null; }); } function moveSCEBlock(blockId, newLayerId, newPosition) { var block = _sceState.blocks.find(function (b) { return Number(b.id) === blockId; }); if (!block) return; var oldLayerId = block.layer_id; var oldStart = block.start_state; var oldEnd = block.end_state; var span = oldEnd - oldStart; block.layer_id = newLayerId; block.start_state = newPosition; block.end_state = newPosition + span; // Push undo scePushUndo({ type: "move_block", blockId: blockId, undo: function () { var b = _sceState.blocks.find(function (bb) { return Number(bb.id) === blockId; }); if (b) { b.layer_id = oldLayerId; b.start_state = oldStart; b.end_state = oldEnd; saveSCEBlock(b); renderSCETimeline(); renderSCEConnections(); } }, redo: function () { var b = _sceState.blocks.find(function (bb) { return Number(bb.id) === blockId; }); if (b) { b.layer_id = newLayerId; b.start_state = newPosition; b.end_state = newPosition + span; saveSCEBlock(b); renderSCETimeline(); renderSCEConnections(); } } }); saveSCEBlock(block); renderSCETimeline(); renderSCEConnections(); } function saveSCEBlock(block) { supabaseRunner .withSuccessHandler(function () {}) .withFailureHandler(function (err) { console.error("[SCE] Failed to save block:", err); }) .sceSaveBlock(JSON.stringify(block)); }
