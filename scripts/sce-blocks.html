// ============================================================ // Story Composition Editor — Block Drag/Drop and Resize // ============================================================ // #sceTimelineBody survives re-renders (only its innerHTML is replaced), // so handlers live at script scope. Re-creating them per render stacks // duplicate listeners and fires one drop handler N times. var sceBlockDropCell = null; var sceBlockDragWired = false; function clearSCEBlockDropTarget() { if (sceBlockDropCell) { sceBlockDropCell.classList.remove("sce-drop-target"); sceBlockDropCell = null; } } function sceDropCellFrom(el) { if (!el) return null; return el.closest(".sce-scene-slot, .sce-layer-unscoped, .sce-layer-cell"); } // The gutter between layer columns is a margin, and margin is not part of // an element's hit-test area. Without this, dropping into the gutter finds // no cell and the move is silently ignored. function sceResolveDropCell(el, clientX, clientY) { var direct = sceDropCellFrom(el); if (direct) return direct; var row = el ? el.closest(".sce-state-row") : null; if (!row) return null; var cells = row.querySelectorAll(".sce-layer-cell"); var nearest = null; for (var i = 0; i < cells.length; i++) { var r = cells[i].getBoundingClientRect(); if (clientX >= r.left && clientX <= r.right) return cells[i]; // Cells are in document order, so the last one ending before the // pointer is the one immediately to its left. if (r.right <= clientX) nearest = cells[i]; } return nearest || cells[0] || null; } function onSCEBlockDragOver(e) { var cell = sceResolveDropCell(e.target, e.clientX, e.clientY); if (!cell) { clearSCEBlockDropTarget(); return; } e.preventDefault(); e.dataTransfer.dropEffect = "copy"; // closest() returns the innermost match, and dragover fires for every // child, so only mark the cell that actually changed. if (cell !== sceBlockDropCell) { clearSCEBlockDropTarget(); sceBlockDropCell = cell; cell.classList.add("sce-drop-target"); } } function onSCEBlockDragLeave(e) { if (sceBlockDropCell && !sceBlockDropCell.contains(e.relatedTarget)) { clearSCEBlockDropTarget(); } } function onSCEBlockDrop(e) { var cell = sceResolveDropCell(e.target, e.clientX, e.clientY); clearSCEBlockDropTarget(); if (!cell) return; e.preventDefault(); var data; try { data = JSON.parse(e.dataTransfer.getData("text/plain")); } catch (err) { return; } // Both drag sources land in the same cell, so the target resolution is shared. var layerId = Number(cell.getAttribute("data-layer-id")); var state = cell.getAttribute("data-state"); if (state === "parking" || state === "later") return; var statePosition = Number(state); var sceneId = cell.getAttribute("data-scene-id") || null; var anchorSceneId = sceneId ? Number(sceneId) : null; if (data.type === "asset") { createSCEBlockFromAsset(data.assetId, layerId, statePosition, anchorSceneId); } else if (data.type === "catalogue") { createSCEBlockFromCatalogue(data.catalogueId, layerId, statePosition, anchorSceneId); } } function onSCEBlockClick(e) { if (e.target.closest(".sce-block")) return; _sceState.selectedBlockId = null; _sceState.selectedAssetId = null; document.querySelectorAll(".sce-block.is-selected").forEach(function (el) { el.classList.remove("is-selected"); }); renderSCEInspector(); } function onSCEBlockDblClick(e) { var blockEl = e.target.closest(".sce-block"); if (!blockEl) return; var blockId = Number(blockEl.getAttribute("data-block-id")); var block = _sceState.blocks.find(function (b) { return Number(b.id) === blockId; }); if (block && block.precomp_composition_id) { openPrecomp(block); } } function wireSCEBlockInteractions() { var timelineBody = document.getElementById("sceTimelineBody"); if (!timelineBody) return; // Allow drops on layer cells and scene slots timelineBody.addEventListener("dragover", onSCEBlockDragOver); timelineBody.addEventListener("dragleave", onSCEBlockDragLeave); timelineBody.addEventListener("drop", onSCEBlockDrop); timelineBody.addEventListener("click", onSCEBlockClick); timelineBody.addEventListener("dblclick", onSCEBlockDblClick); // A drag cancelled with Escape fires no drop, so clear on dragend. document.addEventListener("dragend", clearSCEBlockDropTarget); // Wire block drag (for moving blocks) wireSCEBlockDrag(); } function wireSCEBlockDrag() { // Use mousedown/mousemove/mouseup for block dragging var timelineBody = document.getElementById("sceTimelineBody"); if (!timelineBody) return; // Adds document-level listeners, so only wire once. if (sceBlockDragWired) return; sceBlockDragWired = true; var dragState = null; timelineBody.addEventListener("mousedown", function (e) { var block = e.target.closest(".sce-block"); if (!block) return; if (e.target.closest(".sce-block-connect-node")) return; var blockId = Number(block.getAttribute("data-block-id")); var blockData = _sceState.blocks.find(function (b) { return Number(b.id) === blockId; }); if (!blockData) return; // Check if layer is locked var layer = _sceState.layers.find(function (l) { return Number(l.id) === Number(blockData.layer_id); }); if (layer && layer.is_locked) return; var rect = block.getBoundingClientRect(); dragState = { blockId: blockId, startX: e.clientX, startY: e.clientY, blockEl: block, blockRect: rect, moved: false, ghost: null }; e.preventDefault(); }); document.addEventListener("mousemove", function (e) { if (!dragState) return; if (!dragState.moved) { // First move: create ghost and hide original dragState.moved = true; var ghost = document.createElement("div"); ghost.className = "sce-block-drag-ghost"; ghost.style.cssText = "position:fixed;left:" + dragState.blockRect.left + "px;top:" + dragState.blockRect.top + "px;width:" + dragState.blockRect.width + "px;height:" + dragState.blockRect.height + "px;background:" + (dragState.blockEl.style.getPropertyValue("--layer-color") || "#3b82f6") + ";border-radius:4px;padding:4px 8px;z-index:99999;pointer-events:none;opacity:0.8;display:flex;align-items:center;"; var title = dragState.blockEl.querySelector(".sce-block-title"); ghost.innerHTML = title ? "" + title.innerHTML + "" : ""; document.body.appendChild(ghost); dragState.ghost = ghost; dragState.blockEl.style.opacity = "0.3"; } // Move ghost if (dragState.ghost) { dragState.ghost.style.left = (e.clientX - dragState.startX + dragState.blockRect.left) + "px"; dragState.ghost.style.top = (e.clientY - dragState.startY + dragState.blockRect.top) + "px"; } // Auto-scroll timeline when near edges var timeline = document.getElementById("sceTimeline"); if (timeline) { var wrapRect = timeline.getBoundingClientRect(); var edgeZone = 40; var scrollSpeed = 15; if (e.clientX < wrapRect.left + edgeZone) { timeline.scrollLeft -= scrollSpeed; } else if (e.clientX > wrapRect.right - edgeZone) { timeline.scrollLeft += scrollSpeed; } } }); document.addEventListener("mouseup", function (e) { if (!dragState) return; // Remove ghost if (dragState.ghost) { dragState.ghost.remove(); } dragState.blockEl.style.opacity = ""; if (dragState.moved) { // Find drop target var under = document.elementFromPoint(e.clientX, e.clientY); var cell = sceResolveDropCell(under, e.clientX, e.clientY); if (cell) { var newLayerId = Number(cell.getAttribute("data-layer-id")); var newState = cell.getAttribute("data-state"); if (newState !== "parking" && newState !== "later") { var newPosition = Number(newState); var newSceneId = cell.getAttribute("data-scene-id") || null; moveSCEBlock(dragState.blockId, newLayerId, newPosition, newSceneId ? Number(newSceneId) : null); } } } dragState = null; }); } function moveSCEBlock(blockId, newLayerId, newPosition, newSceneId) { if (sceMutationBlocked("block move")) return; var block = _sceState.blocks.find(function (b) { return Number(b.id) === blockId; }); if (!block) return; var oldLayerId = block.layer_id; var oldStart = block.start_state; var oldEnd = block.end_state; var oldSceneId = block.scene_id || null; var span = oldEnd - oldStart; var anchorScene = newSceneId ? newSceneId : null; block.layer_id = newLayerId; block.scene_id = anchorScene; block.start_state = newPosition; block.end_state = anchorScene ? newPosition : newPosition + span; // Push undo scePushUndo({ type: "move_block", blockId: blockId, undo: function (done) { var b = _sceState.blocks.find(function (bb) { return Number(bb.id) === blockId; }); if (b) { b.layer_id = oldLayerId; b.start_state = oldStart; b.end_state = oldEnd; b.scene_id = oldSceneId; saveSCEBlock(b, function (failed) { renderSCETimeline(); if (done) done(failed); }); } else if (done) done(); }, redo: function (done) { var b = _sceState.blocks.find(function (bb) { return Number(bb.id) === blockId; }); if (b) { b.layer_id = newLayerId; b.start_state = newPosition; b.end_state = anchorScene ? newPosition : newPosition + span; b.scene_id = anchorScene; saveSCEBlock(b, function (failed) { renderSCETimeline(); if (done) done(failed); }); } else if (done) done(); } }); saveSCEBlock(block); renderSCETimeline(); renderSCEConnections(); } function saveSCEBlock(block, done) { supabaseRunner .withSuccessHandler(function () { if (done) done(); }) .withFailureHandler(function (err) { console.error("[SCE] Failed to save block:", err); if (done) done(true); }) .sceSaveBlock(JSON.stringify(block)); }
